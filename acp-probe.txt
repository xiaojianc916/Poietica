8  /// Brings the database up to the current schema version.
9  ///
10  /// Each migration runs inside a transaction together with the row that records
11  /// it, so a failure leaves no half applied schema behind.
12  ///
13  /// # Errors
14  ///
15  /// Fails when a migration statement is rejected.
16  pub(crate) fn migrate(connection: &mut Connection) -> Result<()> {
17      connection.execute_batch(
18          "CREATE TABLE IF NOT EXISTS schema_migrations (
19              version    INTEGER PRIMARY KEY,
20              name       TEXT    NOT NULL,
21              applied_at TEXT    NOT NULL
22           ) STRICT;",
23      )?;
24  
25      let applied: i64 = connection.query_row(
26          "SELECT coalesce(max(version), 0) FROM schema_migrations",
27          [],
28          |row| row.get(0),
29      )?;
30  
31      for (version, name, sql) in MIGRATIONS {
32          if *version <= applied {
33              continue;
34          }
35  
36          let transaction = connection.transaction()?;
37          transaction.execute_batch(sql)?;
38          transaction.execute(
39              "INSERT INTO schema_migrations (version, name, applied_at)
40               VALUES (?1, ?2, datetime('now'))",
41              rusqlite::params![version, name],
42          )?;
43          transaction.commit()?;
44      }
45  
46      Ok(())
47  }
