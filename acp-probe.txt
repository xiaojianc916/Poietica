57  /**
58   * Subscribes to run frames.
59   *
60   * Unsubscribing has to be synchronous for the port, while Tauri's listener
61   * registration is asynchronous, so a handle that arrives after the caller has
62   * already given up is torn down immediately instead of leaking.
63   */
64  export function createAgentEventSource({
65    onListenFailure,
66  }: AgentEventSourceOptions = {}): AgentEventSource {
67    return {
68      listen: (handler) => {
69        let cancelled = false
70        let stop: (() => void) | null = null
71  
72        void import('@tauri-apps/api/event')
73          .then((module) =>
74            module.listen<AgentEventEnvelope>(AGENT_EVENT, (event) => {
75              // The frame is the contract; the envelope is not.
76              handler(event.payload.frame)
77            }),
78          )
79          .then((unlisten) => {
80            if (cancelled) {
81              unlisten()
82              return
83            }
84  
85            stop = unlisten
86          })
87          .catch((error: unknown) => {
88            onListenFailure?.(error)
89          })
90  
91        return () => {
92          cancelled = true
93          stop?.()
94          stop = null
95        }
96      },
97    }
98  }

100  /** Turns a thrown IPC error into the package's error type. */
101  async function call<T>(operation: () => Promise<T>): Promise<T> {
102    try {
103      return await operation()
104    } catch (error) {
105      if (isIpcError(error)) {
106        throw new IpcInvocationError(error)
107      }
108  
109      throw error
110    }
111  }

113  /**
114   * The command half of the port.
115   *
116   * Cancellation ignores the run identifier because a session runs one turn at a
117   * time, so there is exactly one turn to stop. Answering a permission request
118   * is checked natively: an answer naming an option the agent never offered is
119   * refused rather than acted on.
120   */
121  export function createAgentCommandBridge({
122    command,
123    cwd,
124  }: AgentBridgeOptions = {}): AgentCommandBridge {
125    return {
126      prompt: async (request) => {
127        const result = await call(() =>
128          commands.agentPrompt({
129            text: request.text,
130            command: command ?? null,
131            cwd: cwd ?? null,
132          }),
133        )
134  
135        return { runId: result.runId, sessionId: result.sessionId }
136      },
137  
138      cancel: async (_runId) => {
139        await call(() => commands.agentCancel())
140      },
141  
142      resolvePermission: async (requestId, optionId) => {
143        await call(() => commands.agentResolvePermission({ requestId, optionId }))
144      },
145  
146      loadRun: async (runId) => {
147        const snapshot = await call(() => commands.agentLoadRun({ runId, afterSeq: null }))
148  
149        return snapshot.events
150      },
151    }

270  /**
271   * Names the purpose without trusting this build to know every category.
272   *
273   * A category nobody here has heard of is carried as other rather than
274   * dropped: the protocol allows one, and the user should still be able to
275   * change it.
276   */
277  function purposeOf(value: string): AgentConfigPurposeName {
278    /*
279     * Case is a serialisation detail, not a decision the interface should
280     * be at the mercy of. A category the native side spells with a capital
281     * would otherwise land in other, and every row would be filed under a
282     * heading none of them belongs to.
283     */
284    const named = value.toLowerCase()
285  
286    if (named === 'model' || named === 'thought' || named === 'mode') {
287      return named
288    }
289  
290    return 'other'
291  }

298    return {
299      value: native.value,
300      label: native.label,
301      ...(native.detail === null ? {} : { detail: native.detail }),
302    }

43  function group(threads: readonly AssistantThreadSummary[]) {
44    const grouped = new Map<string, AssistantThreadSummary[]>()
45  
46    for (const thread of threads) {
47      const held = grouped.get(thread.group)
48  
49      if (held === undefined) grouped.set(thread.group, [thread])
50      else held.push(thread)
51    }
52  
53    return [...grouped]
54  }
