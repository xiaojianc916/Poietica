461  /// Lists the selectors the running session offers.
462  ///
463  /// The agent reports these when the session is created, so an empty list
464  /// means no session is running yet rather than a session without choices.
465  /// Nothing is invented here: a model, a reasoning level or a mode appears
466  /// in this list only because the agent named it.
467  ///
468  /// # Errors
469  ///
470  /// Fails when the session lock was poisoned or the driver has stopped.
471  #[tauri::command]
472  #[specta::specta]
473  pub async fn agent_config_options(
474      state: State<'_, AgentRuntime>,
475  ) -> AgentCommandResult<Vec<AgentConfigControl>> {
476      // A selector belongs to a session, so asking what is on offer is what
477      // starts one. The alternative was to answer with nothing until the
478      // first prompt, which left the interface showing a model read from a
479      // file that the session, once it existed, might not be following.
480      //
481      // Opening this surface is the user opening the feature, which is not
482      // the same as paying for it at launch: nothing here runs until the
483      // interface asks.
484      let live = ensure_session(&state, None, None).await?;
485  
486      let answer = live.client.selectors().map_err(translate)?;
487      let offered = answer
488          .await
489          .map_err(|_dropped| Error::Internal(NO_ANSWER.to_owned()))?
490          .map_err(translate)?;
491  
492      Ok(offered.into_iter().map(restate).collect())
493  }

495  /// Changes one selector on the running session.
496  ///
497  /// The change applies to the session in flight, so nothing is restarted
498  /// and nothing is written to the agent configuration file. The answer is
499  /// the whole list as the agent reports it afterwards, because one change
500  /// may add or remove another selector.
501  ///
502  /// # Errors
503  ///
504  /// Fails when no session is running, when a turn is in flight, or when
505  /// the agent refuses the value.
506  #[tauri::command]
507  #[specta::specta]
508  pub async fn agent_set_config_option(
509      state: State<'_, AgentRuntime>,
510      request: AgentSelectConfigRequest,
511  ) -> AgentCommandResult<Vec<AgentConfigControl>> {
512      let live = borrow(&state)?.ok_or_else(|| Error::NotFound(NO_SESSION.to_owned()))?;
513  
514      let answer = live
515          .client
516          .select(request.config_id, request.value)
517          .map_err(translate)?;
518      let offered = answer
519          .await
520          .map_err(|_dropped| Error::Internal(NO_ANSWER.to_owned()))?
521          .map_err(translate)?;
522  
523      Ok(offered.into_iter().map(restate).collect())
524  }

527  fn restate(control: ConfigControl) -> AgentConfigControl {
528      AgentConfigControl {
529          id: control.id,
530          label: control.label,
531          detail: control.detail,
532          purpose: match control.purpose {
533              ConfigPurpose::Mode => AgentConfigPurpose::Mode,
534              ConfigPurpose::Model => AgentConfigPurpose::Model,
535              ConfigPurpose::Thought => AgentConfigPurpose::Thought,
536              ConfigPurpose::Other => AgentConfigPurpose::Other,
537          },
538          current: control.current,
539          choices: control
540              .choices
541              .into_iter()
542              .map(|choice| AgentConfigChoice {
543                  value: choice.value,
544                  label: choice.label,
545                  detail: choice.detail,
546              })
547              .collect(),
548      }
549  }
